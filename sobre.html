RNF02 Não-Funcional Desempenho fluido de animação no HTML5 Canvas a 60 FPS sem travamentos.

-- Estrutura do Banco de Dados para LGBT Games CREATE DATABASE IF NOT EXISTS lgbt_games; USE lgbt_games; CREATE TABLE IF NOT EXISTS ranking ( id INT AUTO_INCREMENT PRIMARY KEY, nome_jogador VARCHAR(50) NOT NULL, pontuacao INT NOT NULL, nivel_alcancado INT NOT NULL, data_registro TIMESTAMP DEFAULT CURRENT_TIMESTAMP );
